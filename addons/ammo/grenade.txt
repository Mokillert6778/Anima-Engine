# 手雷（addons/weapon/grenade.txt 使用）
#
# 一颗会弹跳、带引信的手雷：撞墙/地面会反弹（bounce），引信烧完在现场
# 爆炸（explode_damage / explode_radius），爆炸同时走 player / NPC / 可破坏
# 道具的公共伤害通道。

reg_name = grenade

# ---- 飞行 ----
speed = 17              # 抛出初速（米/秒）
damage = 35             # 直接砸到目标的伤害（爆炸伤害另算）
life = 8.0              # 长于引信，避免没炸就消失
gravity = 1.0           # 1 ≈ 9.8 m/s²，手雷必须会掉下来
hit_radius = 0.16
launch_up = 2.4         # 额外抬升，扔出去是抛物线而不是直线

# ---- 爆炸 ----
explode_damage = 120
explode_radius = 5.5
fuse = 2.8              # 2.8 秒后爆炸
bounce = 0.42           # 撞墙/地面的反弹系数

# ---- 外观 ----
visual = model
model = NPC/gman/grenade.mdl
size = 0.32
spin = 9.0              # 飞行时翻滚，能从侧面看出是一颗手雷
